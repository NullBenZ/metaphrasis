cmake_minimum_required(VERSION 3.16)
project(local_graph_converter LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(graph_tool src/main.cpp src/topk.cpp src/leiden.cpp)
target_include_directories(graph_tool PRIVATE include)
if(UNIX)
  target_link_libraries(graph_tool PRIVATE pthread)
endif()
enable_testing()
add_executable(topk_test ../tests/topk_test.cpp src/topk.cpp)
target_include_directories(topk_test PRIVATE include)
if(UNIX)
  target_link_libraries(topk_test PRIVATE pthread)
endif()
add_test(NAME topk_hand_cases COMMAND topk_test)
add_executable(leiden_test ../tests/leiden_test.cpp src/leiden.cpp)
target_include_directories(leiden_test PRIVATE include)
add_test(NAME leiden_hand_cases COMMAND leiden_test)
